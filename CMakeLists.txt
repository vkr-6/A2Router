cmake_minimum_required(VERSION 3.20)
project(A2Router LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
if(MSVC)
    # Portable native binaries do not require a separately installed C++ runtime.
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(a2router_core_test tests/native_core.cpp)
    target_include_directories(a2router_core_test PRIVATE src)
    target_link_libraries(a2router_core_test PRIVATE Threads::Threads)
    add_test(NAME native_core COMMAND a2router_core_test)
endif()
if(WIN32)
    add_library(a2router_engine STATIC src/AudioEngine.cpp)
    target_include_directories(a2router_engine PUBLIC src)
    target_compile_definitions(a2router_engine PRIVATE UNICODE _UNICODE)
    target_link_libraries(a2router_engine PUBLIC ole32 avrt mmdevapi)
    option(A2ROUTER_BUILD_CLI "Build the optional legacy developer console" OFF)
    if(A2ROUTER_BUILD_CLI)
        add_executable(a2router_cli src/main_cli.cpp)
        target_link_libraries(a2router_cli PRIVATE a2router_engine)
    endif()
    if(BUILD_TESTING)
        add_executable(a2router_wasapi_probe tests/windows_probe.cpp)
        target_link_libraries(a2router_wasapi_probe PRIVATE a2router_engine)
        add_test(NAME wasapi_enumeration COMMAND a2router_wasapi_probe)
    endif()
    add_library(a2router_wasapi SHARED src/PythonBridge.cpp)
    target_link_libraries(a2router_wasapi PRIVATE a2router_engine cfgmgr32)
    # A fixed location lets the Python package find both single/multi-config builds.
    set_target_properties(a2router_wasapi PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties(a2router_wasapi PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
    endforeach()
    if(MSVC)
        target_compile_options(a2router_engine PRIVATE /W4 /permissive- /wd4324)
    endif()
endif()
